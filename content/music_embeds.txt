<script>
  function scaleCalendar() {
    const container = document.getElementById('responsive-container');
    const calendar = document.getElementById('calendarScaler');

    const maxWidth = 880; // New cropped width
    const containerWidth = container.offsetWidth;
    const scale = Math.min(1, containerWidth / maxWidth);

    calendar.style.transform = `scale(${scale})`;
  }

  // Initial call
  scaleCalendar();

  // Recalculate on resize
  window.addEventListener('resize', scaleCalendar);
</script>
